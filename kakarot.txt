local function getRoot()
    local char = lp.Character
    if not char then return nil end
    return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
end

local function getAuraCFrame(i)
    local cfg = auraConfigs[i] or auraConfigs[1]
    return CFrame.new(cfg.off.X, cfg.off.Y, cfg.off.Z)
        * CFrame.Angles(math.rad(cfg.rot.X), math.rad(cfg.rot.Y), math.rad(cfg.rot.Z))
end

-- 💜 Tính C0 cho aura phải = lật ngang vị trí aura trái so với nhân vật + GIỮ nguyên hướng nhìn
local function computeRightC0(rightClaw, leftWorldCF)
    local root = getRoot()
    if not root or not leftWorldCF then return nil end

    local rel = root.CFrame:ToObjectSpace(leftWorldCF)          -- vị trí aura trái so với nhân vật
    local p   = rel.Position
    -- lật X → phía đối diện, giữ Y/Z và giữ nguyên hướng của aura
    local desired = root.CFrame * CFrame.new(-p.X, p.Y, p.Z) * rel.Rotation
    return rightClaw.CFrame:ToObjectSpace(desired)
end

local function severHandleWelds(handle)
    if not handle then return end
    local acc = handle.Parent
    local char = lp.Character
    local containers = {handle}
    if acc and acc ~= handle then table.insert(containers, acc) end
    if char then table.insert(containers, char) end

    local seen = {}
    for _, container in ipairs(containers) do
        for _, w in ipairs(container:GetDescendants()) do
            if not seen[w] and (w:IsA("Weld") or w:IsA("WeldConstraint") or w:IsA("Motor6D")) then
                local p0, p1 = w.Part0, w.Part1
                if p0 == handle or p1 == handle then
                    local other = (p0 == handle) and p1 or p0
                    local internal = acc and other and other ~= handle and other:IsDescendantOf(acc)
                    if not internal then
                        seen[w] = true
                        w:Destroy()
                    end
                end
            end
        end
    end
end

local function findAuras()
    auras = {}
    local char = lp.Character
    if not char then return end
    for _, acc in ipairs(char:GetChildren()) do
        if acc:IsA("Accessory") then
            local n = string.lower(acc.Name)
            if n:find("purple", 1, true) and n:find("anime", 1, true) and n:find("aura", 1, true) then
                local handle = acc:FindFirstChild("Handle") or acc:FindFirstChildWhichIsA("BasePart")
                if handle then
                    table.insert(auras, handle)
                    if #auras >= 2 then break end
                end
            end
        end
    end
    print("[Aura] Tìm thấy", #auras, "Purple Anime Aura")
end

local function getClawHands()
    local char = lp.Character
    if not char then return nil, nil end

    local function getHandle(acc)
        if not acc then return nil end
        local h = acc:FindFirstChild("Handle")
        if h and h:IsA("BasePart") then return h end
        for _, v in ipairs(acc:GetDescendants()) do
            if v:IsA("BasePart") then return v end
        end
        return nil
    end

    local left  = getHandle(char:FindFirstChild("Accessory (gears_back_red)"))
    local right = getHandle(char:FindFirstChild("Accessory (gears_back_neon_blue)"))

    if not left or not right then
        for _, acc in ipairs(char:GetChildren()) do
            if acc:IsA("Accessory") then
                local name = string.lower(acc.Name)
                if not left and name:find("gears") and name:find("red") then
                    left = getHandle(acc)
                elseif not right and name:find("gears") and (name:find("blue") or name:find("neon")) then
                    right = getHandle(acc)
                end
            end
        end
    end
    print("[Aura] Claw trái:", left and "OK" or "KHÔNG", "| Claw phải:", right and "OK" or "KHÔNG")
    return left, right
end

    if not handle then return end
    for _, name in ipairs({"EnergyGlow_Light", "EnergyGlow_Particles"}) do

    end
end

    pe.Color = ColorSequence.new(BEAM_COLOR, Color3.fromRGB(230, 150, 255))

    end)
end

local function createBeam(startPos, direction)
    local beamPart = Instance.new("Part")
    beamPart.Anchored, beamPart.CanCollide, beamPart.CanQuery, beamPart.CanTouch = true, false, false, false
    beamPart.Material = Enum.Material.Neon
    beamPart.Color = BEAM_COLOR
    beamPart.Transparency = 0.1
    beamPart.Size = Vector3.new(BEAM_THICK, BEAM_THICK, BEAM_LENGTH)
    beamPart.CFrame = CFrame.lookAt(startPos + direction * (BEAM_LENGTH / 2), startPos + direction * BEAM_LENGTH)
    beamPart.Parent = workspace

    local core = Instance.new("Part")
    core.Anchored, core.CanCollide, core.CanQuery, core.CanTouch = true, false, false, false
    core.Material = Enum.Material.Neon
    core.Color = Color3.fromRGB(255, 220, 255)
    core.Size = Vector3.new(BEAM_THICK * 0.4, BEAM_THICK * 0.4, BEAM_LENGTH)
    core.CFrame = beamPart.CFrame
    core.Parent = workspace

    Debris:AddItem(beamPart, 0.9)
    Debris:AddItem(core, 0.9)
    TweenService:Create(beamPart, TweenInfo.new(0.9, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Transparency = 1, Size = Vector3.new(0.1, 0.1, BEAM_LENGTH * 1.4),
    }):Play()
    TweenService:Create(core, TweenInfo.new(0.9, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Transparency = 1, Size = Vector3.new(0.05, 0.05, BEAM_LENGTH * 1.4),
    }):Play()
end

local function createRadialBurst(centerPos)
    local shock = Instance.new("Part")
    shock.Shape = Enum.PartType.Ball
    shock.Anchored, shock.CanCollide, shock.CanQuery, shock.CanTouch = true, false, false, false
    shock.Material = Enum.Material.Neon
    shock.Color = BEAM_COLOR
    shock.Transparency = 0.4
    shock.Size = Vector3.new(2, 2, 2)
    shock.CFrame = CFrame.new(centerPos)
    shock.Parent = workspace
    Debris:AddItem(shock, 0.7)
    TweenService:Create(shock, TweenInfo.new(0.7, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Transparency = 1, Size = Vector3.new(SHOCKWAVE_SIZE, SHOCKWAVE_SIZE, SHOCKWAVE_SIZE),
    }):Play()

    local golden = math.pi * (3 - math.sqrt(5))
    for i = 1, RADIAL_COUNT do
        local y = 1 - (i - 1) / (RADIAL_COUNT - 1) * 2
        local r = math.sqrt(math.max(0, 1 - y * y))
        local theta = golden * i
        local dir = Vector3.new(math.cos(theta) * r, y, math.sin(theta) * r)

        local len = RADIAL_LENGTH * (0.7 + math.random() * 0.5)
        local beam = Instance.new("Part")
        beam.Anchored, beam.CanCollide, beam.CanQuery, beam.CanTouch = true, false, false, false
        beam.Material = Enum.Material.Neon
        beam.Color = BEAM_COLOR
        beam.Transparency = 0.2
        beam.Size = Vector3.new(0.8, 0.8, len)
        beam.CFrame = CFrame.lookAt(centerPos + dir * (len / 2), centerPos + dir * len)
        beam.Parent = workspace
        Debris:AddItem(beam, 0.6)
        TweenService:Create(beam, TweenInfo.new(0.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Transparency = 1, Size = Vector3.new(0.05, 0.05, len * 1.6),
        }):Play()
    end
end

    trail.LightEmission = 1

    Debris:AddItem(a0, 2)
    Debris:AddItem(a1, 2)
end

local function startShake(duration, power)
    local char = lp.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    if shakeConn then shakeConn:Disconnect() shakeConn = nil end
    local start = tick()
    shakeConn = RunService.RenderStepped:Connect(function()
        local t = tick() - start
        if t >= duration or not hum.Parent then
            hum.CameraOffset = Vector3.zero
            if shakeConn then shakeConn:Disconnect() shakeConn = nil end
            return
        end
        local decay = 1 - (t / duration)
        hum.CameraOffset = Vector3.new(
            (math.random() - 0.5) * 2 * power * decay,
            (math.random() - 0.5) * 2 * power * decay,
            (math.random() - 0.5) * 2 * power * decay
        )
    end)
end

local function putInPalmsAndCharge()
    if isBusy or #auras == 0 then return end
    isBusy = true

    local leftClaw, rightClaw = getClawHands()
    if not leftClaw and not rightClaw then
        print("[Aura] KHÔNG thấy claw gear!")
        isBusy = false
        return
    end

    if spinConn then spinConn:Disconnect() spinConn = nil end

    local leftWorldCF = nil -- vị trí thế giới của aura trái (để mirror sang phải)

    for i, handle in ipairs(auras) do
        if handle and handle.Parent then
            severHandleWelds(handle)
            handle.Anchored = false
            handle.CanCollide = false
            handle.Massless = true
            handle.Transparency = 0

            local target, baseC0

            if #auras >= 2 and i == 1 then
                -- AURA TRÁI → dùng config [1] (đã ổn)
                target = leftClaw or rightClaw
                baseC0 = getAuraCFrame(1)

            elseif #auras >= 2 and i == 2 then
                -- AURA PHẢI → mirror từ aura trái
                target = rightClaw or leftClaw
                if MIRROR_RIGHT_FROM_LEFT and leftWorldCF and (rightClaw or leftClaw) then
                    baseC0 = computeRightC0(rightClaw or leftClaw, leftWorldCF)
                end
                if not baseC0 then
                    baseC0 = getAuraCFrame(2) -- fallback: chỉnh tay
                end

            else
                -- 1 aura duy nhất
                if SINGLE_AURA_TARGET == "left" then
                    target, baseC0 = leftClaw or rightClaw, getAuraCFrame(1)
                elseif SINGLE_AURA_TARGET == "right" then
                    target, baseC0 = rightClaw or leftClaw, getAuraCFrame(2)
                else
                    target = getRoot()
                    local off = CENTER_OFF
                    baseC0 = CFrame.new(off.X, off.Y, off.Z)
                end
            end

            local weld = Instance.new("Weld")
            weld.Name = "AuraPalm_" .. i
            weld.Part0 = target
            weld.Part1 = handle
            weld.C0 = baseC0
            weld.Parent = handle
            auraWelds[i] = weld
            auraBaseC0[i] = baseC0

            -- lưu vị trí thế giới của aura trái để mirror cho aura phải
            if i == 1 and target then
                leftWorldCF = target.CFrame * baseC0
            end

            addChargeGlow(handle)
        end
    end

    isInHand = true
    startShake(CHARGE_TIME, 0.12)
    print("[Aura] Charge " .. CHARGE_TIME .. "s → BẮN! (tốc độ " .. SHOOT_SPEED .. ")")

    local start = tick()
    spinConn = RunService.Heartbeat:Connect(function()
        local t = tick() - start
        if t >= CHARGE_TIME then
            if spinConn then spinConn:Disconnect() spinConn = nil end
            for _, handle in ipairs(auras) do
                removeChargeGlow(handle)
            end
            shootFast()
            return
        end
        local shakeMag = 0.25 * (t / CHARGE_TIME)
        for i, weld in pairs(auraWelds) do
            if weld and weld.Parent and auraBaseC0[i] then
                weld.C0 = auraBaseC0[i]
                    * CFrame.Angles(0, 0, math.rad(t * 720))
                    * CFrame.new((math.random() - 0.5) * shakeMag, (math.random() - 0.5) * shakeMag, 0)
            end
        end
    end)
end

shootFast = function()
    if not isInHand then isBusy = false return end
    local root = getRoot()
    if not root then isBusy = false return end

    local dir = root.CFrame.LookVector
    print("[Aura] 💜 BẮN! Tốc độ = " .. SHOOT_SPEED)

    for _, handle in ipairs(auras) do
        if handle and handle.Parent then
            removeChargeGlow(handle)
            severHandleWelds(handle)
            handle.Anchored = false
            handle.CanCollide = false

            createRadialBurst(handle.Position)
            createBeam(handle.Position, dir)
            addOrbTrail(handle)

            local bv = Instance.new("BodyVelocity")
            bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)
            bv.Velocity = dir * SHOOT_SPEED
            bv.Parent = handle
            Debris:AddItem(bv, BV_LIFETIME)

            task.delay(BV_LIFETIME + 0.1, function()
                if handle and handle.Parent then
                    handle.Transparency = 1
                    pcall(function() handle.Velocity = Vector3.zero end)
                end
            end)
        end
    end

    startShake(0.6, SHAKE_POWER)
    isInHand = false

    task.delay(1.2, function()
        findAuras()
        if #auras > 0 then
            for _, handle in ipairs(auras) do
                if handle then handle.Transparency = 0 end
            end
            hideUnderCharacter()
        end
        isBusy = false
        print("[Aura] Đã quay lại dưới nhân vật 20 stud")
    end)
end

local function novaShake(duration, power)
    local char = lp6.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local start, conn = tick(), nil
    conn = RS6.RenderStepped:Connect(function()
        local t = tick() - start
        if t >= duration or not hum.Parent then
            hum.CameraOffset = Vector3.zero
            conn:Disconnect()
            return
        end
        local d = 1 - t / duration
        hum.CameraOffset = Vector3.new((math.random()-.5)*2*power*d, (math.random()-.5)*2*power*d, (math.random()-.5)*2*power*d)
    end)
end

local function newNeon(color, transparency)
    local p = Instance.new("Part")
    p.Anchored = true
    p.CanCollide = false
    p.CanQuery = false
    p.CanTouch = false
    p.Material = Enum.Material.Neon

    p.Parent = workspace
    return p
end

-- ⚡ TẠO ĐOẠN SÉT (nối 2 điểm bằng các khúc gãy ngẫu nhiên — sét giật zigzag)
local function createLightning(fromPos, toPos, color, thick)
    local dist = (toPos - fromPos).Magnitude
    local segments = math.clamp(math.floor(dist / 4), 3, 14)
    local prevPoint = fromPos
    for i = 1, segments do
        local alpha = i / segments
        local point = fromPos:Lerp(toPos, alpha)
        if i < segments then
            point += Vector3.new(
                (math.random() - 0.5) * 6,
                (math.random() - 0.5) * 6,
                (math.random() - 0.5) * 6
            )
        end
        local mid = (prevPoint + point) / 2
        local segLen = (point - prevPoint).Magnitude
        local seg = newNeon(color, 0.1)
        seg.Size = Vector3.new(thick or 0.35, thick or 0.35, segLen)
        seg.CFrame = CFrame.lookAt(mid, point)
        seg.Parent = workspace
        DB6:AddItem(seg, 0.15 + math.random() * 0.1)
        TS6:Create(seg, TweenInfo.new(0.25), {Transparency = 1}):Play()
        prevPoint = point
    end
end

-- 🌑 VẾT NỨT ĐẤT tỏa ra từ điểm nổ
local function createGroundCracks(pos)
    for i = 1, CRACK_COUNT do
        local ang = (i / CRACK_COUNT) * math.pi * 2 + math.random() * 0.3
        local len = CRACK_LENGTH * (0.6 + math.random() * 0.6)
        local dir = Vector3.new(math.cos(ang), 0, math.sin(ang))

        -- 1 vết nứt = nhiều khúc nhỏ nối tiếp, càng xa càng mỏng + mờ
        local steps = 6
        local prev = pos
        for s = 1, steps do
            local alpha = s / steps
            local point = pos + dir * (len * alpha)
                + Vector3.new((math.random()-0.5)*2, 0, (math.random()-0.5)*2)
            local segLen = (point - prev).Magnitude
            local crack = newNeon(COLOR_CRACK, 0.1)
            crack.Size = Vector3.new(1.2 * (1 - alpha * 0.7), 0.15, segLen)
            crack.CFrame = CFrame.lookAt((prev + point) / 2 - Vector3.new(0, 1.4, 0), point)
            crack.Parent = workspace
            -- nứt hiện ra dần theo khoảng cách (ripple effect)
            DB6:AddItem(crack, 4 + alpha * 2)
            TS6:Create(crack, TweenInfo.new(3 + alpha * 2), {Transparency = 1}):Play()
            prev = point
        end
    end
    -- hố lõm trung tâm
    local hole = newNeon(Color3.fromRGB(20, 0, 0), 0.2)
    hole.Shape = Enum.PartType.Cylinder
    hole.Size = Vector3.new(0.3, 14, 14)
    hole.CFrame = CFrame.new(pos - Vector3.new(0, 1.3, 0)) * CFrame.Angles(0, 0, math.rad(90))
    hole.Parent = workspace
    DB6:AddItem(hole, 5)
    TS6:Create(hole, TweenInfo.new(4.5), {Transparency = 1}):Play()
end

    spart.Position = pos
    spart.Parent = workspace

    snd.PlaybackSpeed = 0.8
    snd.Parent = spart
    snd:Play()
    DB6:AddItem(spart, 5)

    local light = Instance.new("PointLight")
    light.Color = COLOR_MAIN
    light.Brightness = 15
    light.Range = 180
    light.Parent = spart
    DB6:AddItem(light, 1)

    -- 🌑 NỨT ĐẤT LAN RA TRƯỚC TIÊN
    createGroundCracks(pos)

    -- sóng xung kích đỏ
    local ball = newNeon(COLOR_MAIN, 0.1)
    ball.Shape = Enum.PartType.Ball
    ball.Size = Vector3.new(5, 5, 5)
    ball.Position = pos
    DB6:AddItem(ball, 1.4)
    TS6:Create(ball, TweenInfo.new(1.4, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Transparency = 1,
        Size = Vector3.new(EXPLOSION_SIZE, EXPLOSION_SIZE, EXPLOSION_SIZE),
    }):Play()

    -- lõi trắng đỏ
    local core = newNeon(COLOR_BRIGHT, 0)
    core.Shape = Enum.PartType.Ball
    core.Size = Vector3.new(4, 4, 4)
    core.Position = pos
    DB6:AddItem(core, 0.7)
    TS6:Create(core, TweenInfo.new(0.7, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
        Transparency = 1,
        Size = Vector3.new(EXPLOSION_SIZE * 0.55, EXPLOSION_SIZE * 0.55, EXPLOSION_SIZE * 0.55),
    }):Play()

    -- tia đỏ tỏa mọi hướng
    local golden = math.pi * (3 - math.sqrt(5))
    for i = 1, BEAM_COUNT do
        local y = 1 - (i - 1) / (BEAM_COUNT - 1) * 2
        local r = math.sqrt(math.max(0, 1 - y * y))
        local theta = golden * i
        local dir = Vector3.new(math.cos(theta) * r, y, math.sin(theta) * r)
        local len = 45 + math.random() * 35
        local beam = newNeon(COLOR_MAIN, 0.1)
        beam.Size = Vector3.new(1.8, 1.8, len)
        beam.CFrame = CFrame.lookAt(pos + dir * (len / 2), pos + dir * len)
        DB6:AddItem(beam, 1.1)
        TS6:Create(beam, TweenInfo.new(1.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Transparency = 1,
            Size = Vector3.new(0.05, 0.05, len * 1.6),
        }):Play()
    end

    -- ⚡ SÉT NỔ TUNG khi nổ (8 cú giật liên tiếp quanh điểm nổ)
    task.spawn(function()
        for i = 1, 8 do
            local dir2 = Vector3.new(math.random(-40, 40), 0, math.random(-40, 40))
            createLightning(pos, pos + dir2, COLOR_BRIGHT, 0.6)
            createLightning(pos + Vector3.new(0, 30, 0), pos + dir2, COLOR_MAIN, 0.4)
            task.wait(0.08)
        end
    end)

    -- vòng đỏ lan mặt đất
    local ring = newNeon(COLOR_MAIN, 0.3)
    ring.Shape = Enum.PartType.Cylinder
    ring.Size = Vector3.new(0.4, 8, 8)
    ring.CFrame = CFrame.new(pos - Vector3.new(0, 1.5, 0)) * CFrame.Angles(0, 0, math.rad(90))
    DB6:AddItem(ring, 1.8)
    TS6:Create(ring, TweenInfo.new(1.8, Enum.EasingStyle.Quad), {
        Transparency = 1,
        Size = Vector3.new(0.1, EXPLOSION_SIZE * 1.8, EXPLOSION_SIZE * 1.8),
    }):Play()

    -- mảnh đỏ văng
    for i = 1, 8 do -- lag fix
        local chunk = newNeon(i % 2 == 0 and COLOR_MAIN or COLOR_BRIGHT, 0)
        local s = 0.5 + math.random() * 1.5
        chunk.Size = Vector3.new(s, s, s)
        chunk.Position = pos + Vector3.new(math.random(-4, 4), 1, math.random(-4, 4))
        local vel = Instance.new("BodyVelocity")
        vel.MaxForce = Vector3.new(1e9, 1e9, 1e9)
        vel.Velocity = Vector3.new(math.random(-60, 60), math.random(70, 140), math.random(-60, 60))
        vel.Parent = chunk
        DB6:AddItem(vel, 0.15)
        DB6:AddItem(chunk, 1.4)
        TS6:Create(chunk, TweenInfo.new(1.4), {Transparency = 1}):Play()
    end

    novaShake(1.6, SHAKE_POWER)
    print("[Meteor] 💥 NỔ + SÉT + NỨT ĐẤT!")
end

-- 🌑 SEQUENCE CHÍNH
local function dropRedMeteor()
    if meteorBusy then return end
    meteorBusy = true

    -- PHASE 1: CẦU LƠ LỬNG THEO CHUỘT + SÉT GIẬT QUANH CẦU
    local sphere = newNeon(COLOR_MAIN, 0.05)
    sphere.Name = "MeteorCore"
    sphere.Shape = Enum.PartType.Ball
    sphere.Size = Vector3.new(2, 2, 2)

    local shell = newNeon(COLOR_BRIGHT, 0.6)
    shell.Shape = Enum.PartType.Ball
    shell.Size = Vector3.new(3, 3, 3)

    local inner = newNeon(COLOR_BRIGHT, 0)   -- lõi trắng như video
    inner.Shape = Enum.PartType.Ball
    inner.Size = Vector3.new(1, 1, 1)

    local light = Instance.new("PointLight")
    light.Color = COLOR_MAIN
    light.Range = 50
    light.Brightness = 5
    light.Parent = sphere

    local pe = Instance.new("ParticleEmitter")
    pe.Color = ColorSequence.new(COLOR_MAIN, COLOR_BRIGHT)
    pe.LightEmission = 1
    pe.Size = NumberSequence.new(1.8, 0)
    pe.Transparency = NumberSequence.new(0.1, 1)
    pe.Lifetime = NumberRange.new(0.4, 0.9)
    pe.Rate = 60  -- lag fix
    pe.Speed = NumberRange.new(5, 14)
    pe.SpreadAngle = Vector2.new(180, 180)
    pe.Parent = sphere

    chargeLightningOn = true   -- ⚡ bật sét
    novaShake(CHARGE_TIME, 0.8)

    local start = tick()
    local target = nil
    while tick() - start < CHARGE_TIME do
        local t = tick() - start
        local prog = t / CHARGE_TIME
        local hitPos = mouse6.Hit and mouse6.Hit.Position
        if hitPos then target = hitPos end
        if target then
            local hoverPos = target + Vector3.new(0, HOVER_HEIGHT, 0)
            local pulse = 1 + math.sin(t * 16) * 0.06
            local size = (2 + (SPHERE_SIZE - 2) * prog) * pulse
            sphere.Size = Vector3.new(size, size, size)
            shell.Size = Vector3.new(size * 1.25, size * 1.25, size * 1.25)
            inner.Size = Vector3.new(size * 0.4, size * 0.4, size * 0.4)
            sphere.CFrame = CFrame.new(hoverPos)
            shell.CFrame = CFrame.new(hoverPos)
            inner.CFrame = CFrame.new(hoverPos) * CFrame.Angles(t, t * 1.3, 0)
            light.Brightness = 5 + 10 * prog
        end
        task.wait()
    end
    chargeLightningOn = false   -- ⚡ tắt sét charge

    -- PHASE 2: RƠI XUỐNG + VẸT SÉT ĐUÔI CẦU
    if not target then
        sphere:Destroy()
        shell:Destroy()
        inner:Destroy()
        meteorBusy = false
        return
    end

    print("[Meteor] ⬇️ RƠI + SÉT ĐUÔI!")
    local fallPos = sphere.Position
    local params = RaycastParams.new()
    local char = lp6.Character
    if char then
        params.FilterDescendantsInstances = {char}
    end
    params.FilterType = Enum.RaycastFilterType.Exclude

    local trailTimer = 0
    local fallConn = RS6.Heartbeat:Connect(function(dt)
        fallPos = fallPos - Vector3.new(0, FALL_SPEED * dt, 0)
        sphere.CFrame = CFrame.new(fallPos)
        shell.CFrame = CFrame.new(fallPos)
        inner.CFrame = CFrame.new(fallPos)
        -- ⚡ vệt sét đỏ đuôi theo cầu khi rơi
        trailTimer += dt
        if trailTimer > 0.06 then
            trailTimer = 0
            createLightning(fallPos, fallPos + Vector3.new(math.random(-8, 8), 18, math.random(-8, 8)), COLOR_MAIN, 0.35)
        end
    end)

    local impact = nil
    local fallStart = tick()
    while tick() - fallStart < 4 do
        local result = workspace:Raycast(fallPos, Vector3.new(0, -14, 0), params)
        if result then
            impact = result.Position
            break
        end
        task.wait()
    end
    fallConn:Disconnect()
    sphere:Destroy()
    shell:Destroy()
    inner:Destroy()

    -- PHASE 3: 💥 NỔ + SÉT + NỨT ĐẤT
    redExplosion(impact or fallPos)

    task.wait(COOLDOWN)
    meteorBusy = false
    print("[Meteor] Sẵn sàng — bấm U lần nữa!")
end

local function easeOutQuad(t)   t = math.clamp(t,0,1) return 1-(1-t)*(1-t) end
local function easeInQuad(t)    t = math.clamp(t,0,1) return t*t end
local function easeInOutQuad(t) t = math.clamp(t,0,1)
    if t < 0.5 then return 2*t*t end
    return 1-((-2*t+2)^2)/2
end
local function easeOutQuart(t)  t = math.clamp(t,0,1) return 1-(1-t)^4 end

local function align(Part0, Part1)
    
    local att0 = Instance.new("Attachment")

    local att1 = Instance.new("Attachment")

                while twait(2.9) and Part0 and c do
                    hide = #Part0:GetConnectedParts() == 1
                    twait(0.1)
                    hide = false
                end
            end)
        end
        
        local rot = rad(0.05)
        local con0, con1 = nil, nil
        local _rvTick = 0
        con0 = stepped:Connect(function()
            if not (Part0 and Part1) then return con0:Disconnect() and con1:Disconnect() end
            _rvTick = _rvTick + 1
            if _rvTick % 2 == 0 then -- lag fix: RotVelocity half rate
                Part0.RotVelocity = Part1.RotVelocity
            end
        end)
        local lastpos, vel = Part0.Position, Part0.Velocity
        con1 = heartbeat:Connect(function(delta)
            if not (Part0 and Part1 and att1) then return con0:Disconnect() and con1:Disconnect() end
            if (not Part0.Anchored) and (Part0.ReceiveAge == 0) then
                if lostpart == Part0 then
                    lostpart = nil
                end
                local newcf = Part1.CFrame * att1.CFrame
                local vel = (newcf.Position - lastpos) / delta
                Part0.Velocity = getNetlessVelocity(vel)
                if vel.Magnitude < 1 then
                    rot = -rot
                    newcf *= angles(0, 0, rot)
                end
                lastpos = newcf.Position
                if lostpart and (Part0 == reclaim) then
                    newcf = lostpart.CFrame
                elseif hide then
                    newcf += v3_hide
                end
                if novoid and (newcf.Y < ws.FallenPartsDestroyHeight + 0.1) then
                    newcf += v3(0, ws.FallenPartsDestroyHeight + 0.1 - newcf.Y, 0)
                end
                Part0.CFrame = newcf
            elseif (not Part0.Anchored) and (abs(Part0.Velocity.X) < 45) and (abs(Part0.Velocity.Y) < 25) and (abs(Part0.Velocity.Z) < 45) then
                lostpart = Part0
            end
        end)
    
    else
        
        Part0.CustomPhysicalProperties = physp
        if (alignmode == 1) or (alignmode == 2) then
            local ape = Instance.new("AlignPosition")

            apd.Attachment0, apd.Attachment1, apd.Name = att0, att1, "AlignPositionRfalse"
            apd.Parent = att0
        end
        
        local ao = Instance.new("AlignOrientation")
        ao.MaxAngularVelocity, ao.MaxTorque, ao.Responsiveness = inf, inf, inf
        ao.PrimaryAxisOnly, ao.ReactionTorqueEnabled, ao.RigidityEnabled = false, false, false
        ao.Attachment0, ao.Attachment1 = att0, att1
        ao.Parent = att0
        
        local con1 = nil
        local lastpos = Part0.Position
        con1 = heartbeat:Connect(function(delta)
            if not (Part0 and Part1) then return con1:Disconnect() end
            Part0.Velocity = getNetlessVelocity((Part1.Position - lastpos) / delta)
            lastpos = Part1.Position
        end)
    
    end

    att0:GetPropertyChangedSignal("Parent"):Connect(function()
        Part0 = att0.Parent
        if not Part0:IsA("BasePart") then
            att0 = nil
            if lostpart == Part0 then
                lostpart = nil
            end
            Part0 = nil

    att1:GetPropertyChangedSignal("Parent"):Connect(function()
        Part1 = att1.Parent
        if not Part1:IsA("BasePart") then
            att1 = nil
            Part1 = nil
        end
    end)

end

function clerp(a,b,t) 
local qa = {QuaternionFromCFrame(a)}
local qb = {QuaternionFromCFrame(b)} 
local ax, ay, az = a.x, a.y, a.z 
local bx, by, bz = b.x, b.y, b.z
local _t = 1-t
return QuaternionToCFrame(_t*ax + t*bx, _t*ay + t*by, _t*az + t*bz,QuaternionSlerp(qa, qb, t)) 
end 
 
function QuaternionFromCFrame(cf) 
local mx, my, mz, m00, m01, m02, m10, m11, m12, m20, m21, m22 = cf:components() 
local trace = m00 + m11 + m22 
if trace > 0 then 
local s = math.sqrt(1 + trace) 
local recip = 0.5/s 
return (m21-m12)*recip, (m02-m20)*recip, (m10-m01)*recip, s*0.5 
else 
local i = 0 
if m11 > m00 then
i = 1
end
if m22 > (i == 0 and m00 or m11) then 
i = 2 
end 
if i == 0 then 
local s = math.sqrt(m00-m11-m22+1) 
local recip = 0.5/s 
return 0.5*s, (m10+m01)*recip, (m20+m02)*recip, (m21-m12)*recip 
elseif i == 1 then 
local s = math.sqrt(m11-m22-m00+1) 
local recip = 0.5/s 
return (m01+m10)*recip, 0.5*s, (m21+m12)*recip, (m02-m20)*recip 
elseif i == 2 then 
local s = math.sqrt(m22-m00-m11+1) 
local recip = 0.5/s return (m02+m20)*recip, (m12+m21)*recip, 0.5*s, (m10-m01)*recip 
end 
end 
end
 
function QuaternionToCFrame(px, py, pz, x, y, z, w) 
local xs, ys, zs = x + x, y + y, z + z 
local wx, wy, wz = w*xs, w*ys, w*zs 
local xx = x*xs 
local xy = x*ys 
local xz = x*zs 
local yy = y*ys 
local yz = y*zs 
local zz = z*zs 
return CFrame.new(px, py, pz,1-(yy+zz), xy - wz, xz + wy,xy + wz, 1-(xx+zz), yz - wx, xz - wy, yz + wx, 1-(xx+yy)) 
end
 
function QuaternionSlerp(a, b, t) 
local cosTheta = a[1]*b[1] + a[2]*b[2] + a[3]*b[3] + a[4]*b[4] 
local startInterp, finishInterp; 
if cosTheta >= 0.0001 then 
if (1 - cosTheta) > 0.0001 then 
local theta = math.acos(cosTheta) 
local invSinTheta = 1/math.sin(theta) 
startInterp = math.sin((1-t)*theta)*invSinTheta 
finishInterp = math.sin(t*theta)*invSinTheta  
else 
startInterp = 1-t 
finishInterp = t 
end 
else 
if (1+cosTheta) > 0.0001 then 
local theta = math.acos(-cosTheta) 
local invSinTheta = 1/math.sin(theta) 
startInterp = math.sin((t-1)*theta)*invSinTheta 
finishInterp = math.sin(t*theta)*invSinTheta 
else 
startInterp = t-1 
finishInterp = t 
end 
end 
return a[1]*startInterp + b[1]*finishInterp, a[2]*startInterp + b[2]*finishInterp, a[3]*startInterp + b[3]*finishInterp, a[4]*startInterp + b[4]*finishInterp 
end

    local function weldBetween(a, b)
        local weldd = Instance.new("ManualWeld")
        weldd.Part0 = a
        weldd.Part1 = b
        weldd.C0 = CFrame.new()
        weldd.C1 = b.CFrame:inverse() * a.CFrame
        weldd.Parent = a
        return weldd
    end
    
    it=Instance.new
    
    function nooutline(part)
        part.TopSurface,part.BottomSurface,part.LeftSurface,part.RightSurface,part.FrontSurface,part.BackSurface = 10,10,10,10,10,10
    end
    
    function part(formfactor,parent,material,reflectance,transparency,brickcolor,name,size)
        local fp=it("Part")
        fp.formFactor=formfactor
        fp.Parent=parent
        fp.Reflectance=reflectance
        fp.Transparency=transparency
        fp.CanCollide=false
        fp.Locked=true
        fp.BrickColor=BrickColor.new(tostring(brickcolor))
        fp.Name=name
        fp.Size=size
        fp.Position=Character.Torso.Position
        nooutline(fp)
        fp.Material=material
        fp:BreakJoints()
        return fp
    end
    
    function mesh(Mesh,part,meshtype,meshid,offset,scale)
        local mesh=it(Mesh)
        mesh.Parent=part
        if Mesh=="SpecialMesh" then
            mesh.MeshType=meshtype
            mesh.MeshId=meshid
        end
        mesh.Offset=offset
        mesh.Scale=scale
        return mesh
    end
    
    function weld(parent,part0,part1,c0,c1)
        local weld=it("Weld")
        weld.Parent=parent
        weld.Part0=part0
        weld.Part1=part1
        weld.C0=c0
        weld.C1=c1
        return weld
    end

				local att = h and h:FindFirstChild("att1_Handle")
				if att then list[#list + 1] = att end
			end
		end
		return list
	end


	local t0    = os.clock()
	local tips  = collectTips()
	while #tips == 0 and (os.clock() - t0) < MAX_WAIT do
		task.wait(0.2)
		tips = collectTips()
	end

	local cubes = collectCubes()
	while #cubes < #tips * PER_TIP and (os.clock() - t0) < MAX_WAIT do
		task.wait(0.2)
		cubes = collectCubes()
	end

	if #tips == 0 or #cubes == 0 then
		return
	end

	if RANDOM_ORDER then
		for i = #cubes, 2, -1 do
			local j = math.random(i)
			cubes[i], cubes[j] = cubes[j], cubes[i]
		end
	end

	local idx, done = 0, 0
	for _, tip in ipairs(tips) do
		for k = 1, PER_TIP do
			idx = idx + 1
			local att = cubes[idx]
			if not att then break end
			att.Parent   = tip
			att.Rotation = ROTATION
			att.Position = OFFSET
			done = done + 1
		end
	end
end)

		if handleweld then

				CFrame.new(0, 1.2 * rise, -0.8 * rise)
				* CFrame.Angles(math.rad(35 * rise), math.rad(10 * rise), math.rad(0)), 0.3)
		end

				* CFrame.Angles(math.rad(-15 * rise), math.rad(20 + 70 * rise), math.rad(25 * rise)), 0.32)
		end
		if Handle2weld then
			local ang = math.rad(90 * rise) + math.pi
			Handle2weld.C0 = clerp(Handle2weld.C0,
				CFrame.new(math.sin(ang) * 3.4, 0.3 + 1.3 * rise, math.cos(ang) * 2.2)
				* CFrame.Angles(math.rad(-15 * rise), math.rad(180 - 20 - 70 * rise), math.rad(-25 * rise)), 0.32)

			math.rad(-25 + armR2 * 12),
			math.rad(55 + armR * 18)
		)
		-- Tay trái: duỗi + vẫy (±30° / ±0.25 stud)
		LW.C0 = CFrame.new(
			-1.35 + armL * 0.1,
			0.25 + armL * 0.25 + bob * 0.12,
			0.15 + armL2 * 0.15
		) * CFrame.Angles(
			math.rad(-35 + armL * 32),
			math.rad(15 + armL2 * 10),
			math.rad(-40 + armL * 22)
		)
		-- Chân phải: co gối rõ (±0.3 stud / ±22°)

		end
	end

	if moveConn then moveConn:Disconnect() end

	so("rbxassetid://847061203", Character, 0.6, 1.1)
	for i = 0, 1, 0.035 do

		local t = i * i * (3 - 2 * i)

		LH.C0 = clerp(LH.C0, CFrame.new(-1, -1, 0) * CFrame.Angles(0, math.rad(-90), 0), 0.38)
		-- Claws thu về idle
		if handleweld then handleweld.C0 = clerp(handleweld.C0, CFrame.new(0, 0, 0), 0.3) end
		if Handleweld then
			Handleweld.C0 = clerp(Handleweld.C0, CFrame.new(0, 0, 0) * CFrame.Angles(0, 0, 0), 0.3)

function Hai()
    attack = true
    Humanoid.WalkSpeed = 0

    local laughSound = Instance.new("Sound")
    laughSound.SoundId = "rbxassetid://4611186611"
    laughSound.Volume = 1
    laughSound.Parent = Torso
    laughSound:Play()

    local sine = 0
    local phase = "laugh"
    local timer = 0
    local animSpeed = 1.0
    local transitionAlpha = 0.06  -- мягкий lerp для переходов между фазами
    local normalAlpha = 0.1        -- обычный lerp внутри фазы
    local currentAlpha = transitionAlpha

    local laughConn
    laughConn = game:GetService("RunService").RenderStepped:Connect(function(dt)
        timer = timer + dt
        sine = sine + dt

        -- Плавное нарастание lerp после смены фазы
        if currentAlpha < normalAlpha then
            currentAlpha = math.min(currentAlpha + dt * 0.15, normalAlpha)
        end

        local prevPhase = phase

        if phase == "laugh" and timer >= 1 then
            phase = "talk"
            timer = 0
            currentAlpha = transitionAlpha
        elseif phase == "talk" and timer >= 4 then
            phase = "done"
        end

        if phase == "done" then
            laughConn:Disconnect()
            attack = false
            Humanoid.WalkSpeed = 16
            return
        end

        -- Плавный переход внутри talk (на 2 секунде)
        if phase == "talk" and math.abs(timer - 2) < 0.05 then
            currentAlpha = transitionAlpha
        end

        local s12 = sine * 5
        local s24 = sine * 2.5

        local pSwayX  = -0.04 * math.sin(s24)
        local pSwayY  = 0.04 * math.sin(s12)
        local pSwayZ  = 0.05 * math.cos(s12)
        local pRotX   = math.rad(-2.5 * math.sin(s12))
        local pRotY   = math.rad(-2.5 * math.sin(s24))
        local pShake  = 2.5 * math.sin(s12)
        local pShake24 = 2.5 * math.sin(s24)

        local alpha = currentAlpha / animSpeed
        local bodyRot
        local headRot

        if phase == "laugh" then
            local cosVal = math.cos(sine / 2)
            local sinVal = math.sin(sine / 2)
            local lift = (1 - cosVal) / 2
            local torsoLift = lift * 0.5

            bodyRot = angles(math.rad(-10) + pRotX, pRotY, math.rad(0))

            RootJoint.C0 = clerp(
                RootJoint.C0,
                RootCF * cf(pSwayX, pSwayY + torsoLift, -0.05 + 0.05 * cosVal + pSwayZ) * bodyRot,
                alpha
            )

            headRot = angles(math.rad(15 - 1 * math.sin(s12)), math.rad(0), math.rad(0))
            Torso.Neck.C0 = clerp(Torso.Neck.C0, necko * cf(0, 0, 0) * headRot, alpha)

            RW.C0 = clerp(
                RW.C0,
                cf(1.5, 0.5 + 0.05 * sinVal + 0.04 * math.sin(s12), 0)
                    * angles(math.rad(35 + pShake), math.rad(pShake24), math.rad(-5 + pShake)),
                alpha
            )

            LW.C0 = clerp(
                LW.C0,
                cf(-1.5, 0.5 + 0.05 * sinVal - 0.04 * math.sin(s12), 0)
                    * angles(math.rad(-12 - pShake), math.rad(-pShake24), math.rad(5 - pShake)),
                alpha
            )

            RH.C0 = clerp(
                RH.C0,
                cf(1, -1 - 0.05 * cosVal + 0.06 * math.sin(s24) - 0.05 * math.cos(s12), -0.01)
                    * angles(math.rad(-2.5 * math.sin(s12)), math.rad(80), math.rad(0))
                    * angles(math.rad(-5 - 2 - 2.5 * math.sin(s24)), math.rad(0), math.rad(-20)),
                alpha
            )

            LH.C0 = clerp(
                LH.C0,
                cf(-1, -1 - 0.05 * cosVal - 0.05 * math.cos(s12), -0.01)
                    * angles(math.rad(0), math.rad(-80), math.rad(0))
                    * angles(math.rad(-5 - 8), math.rad(0), math.rad(20)),
                alpha
            )

        elseif phase == "talk" then
            if timer < 2 then
                bodyRot = angles(math.rad(-10) + pRotX, pRotY, math.rad(0))

                RootJoint.C0 = clerp(
                    RootJoint.C0,
                    RootCF * cf(pSwayX, pSwayY, -0.05 + pSwayZ) * bodyRot,
                    alpha
                )

                headRot = angles(math.rad(10 - 1 * math.sin(s12)), math.rad(0), math.rad(0))
                Torso.Neck.C0 = clerp(Torso.Neck.C0, necko * cf(0, 0, 0) * headRot, alpha)

                RW.C0 = clerp(
                    RW.C0,
                    cf(1.5, 0.5 + 0.04 * math.sin(s12), 0.5)
                        * angles(math.rad(10), math.rad(0), math.rad(50 + pShake))
                        * angles(math.rad(0), math.rad(-90), math.rad(0)),
                    alpha
                )

                LW.C0 = clerp(
                    LW.C0,
                    cf(-1.5, 0.5 - 0.04 * math.sin(s12), 0.5)
                        * angles(math.rad(10), math.rad(0), math.rad(-50 - pShake))
                        * angles(math.rad(0), math.rad(90), math.rad(0)),
                    alpha
                )

                RH.C0 = clerp(
                    RH.C0,
                    cf(1, -1 + 0.06 * math.sin(s24) - 0.05 * math.cos(s12), -0.01)
                        * angles(math.rad(-2.5 * math.sin(s12)), math.rad(80), math.rad(0))
                        * angles(math.rad(-5 - 2 - 2.5 * math.sin(s24)), math.rad(0), math.rad(-20)),
                    alpha
                )

                LH.C0 = clerp(
                    LH.C0,
                    cf(-1, -1 - 0.05 * math.cos(s12), -0.01)
                        * angles(math.rad(0), math.rad(-80), math.rad(0))
                        * angles(math.rad(-5 - 8), math.rad(0), math.rad(20)),
                    alpha
                )
            else
                bodyRot = angles(math.rad(10) + pRotX, pRotY, math.rad(0))

                RootJoint.C0 = clerp(
                    RootJoint.C0,
                    RootCF * cf(pSwayX, pSwayY, -0.05 + pSwayZ) * bodyRot,
                    alpha
                )

                headRot = angles(math.rad(-10 - 1 * math.sin(s12)), math.rad(0), math.rad(0))
                Torso.Neck.C0 = clerp(Torso.Neck.C0, necko * cf(0, 0, 0) * headRot, alpha)

                RW.C0 = clerp(
                    RW.C0,
                    cf(1.5, 0.5 + 0.04 * math.sin(s12), 0.5)
                        * angles(math.rad(10), math.rad(0), math.rad(50 + pShake))
                        * angles(math.rad(0), math.rad(-90), math.rad(0)),
                    alpha
                )

                LW.C0 = clerp(
                    LW.C0,
                    cf(-1.5, 0.5 - 0.04 * math.sin(s12), 0.5)
                        * angles(math.rad(10), math.rad(0), math.rad(-50 - pShake))
                        * angles(math.rad(0), math.rad(90), math.rad(0)),
                    alpha
                )

                RH.C0 = clerp(
                    RH.C0,
                    cf(1, -1 + 0.06 * math.sin(s24) - 0.05 * math.cos(s12), -0.01)
                        * angles(math.rad(-2.5 * math.sin(s12)), math.rad(80), math.rad(0))
                        * angles(math.rad(-5 - 2 - 2.5 * math.sin(s24)), math.rad(0), math.rad(20)),
                    alpha
                )

                LH.C0 = clerp(
                    LH.C0,
                    cf(-1, -1 - 0.05 * math.cos(s12), -0.01)
                        * angles(math.rad(0), math.rad(-80), math.rad(0))
                        * angles(math.rad(-5 - 8), math.rad(0), math.rad(-20)),
                    alpha
                )
            end
        end

        -- handleweld: lerp вместо жёсткой установки
        if headRot then
            handleweld.C0 = clerp(handleweld.C0, cf(0, 0, 0) * headRot, alpha)
        end

        -- Handleweld и Handle2weld: lerp вместо жёсткой установки
        if bodyRot then
            Handleweld.C0 = clerp(Handleweld.C0, cf(0, 0, 0) * bodyRot, alpha)
            Handle2weld.C0 = clerp(Handle2weld.C0, cf(0, 0, 0) * bodyRot, alpha)
        end
    end)
end


function cabun()

	if anim then anim.Disabled = true end

	local ran, err = pcall(function()

		local TS2 = game:GetService("TweenService")

		local FLY_TIME        = 1
		local CHARGE_TIME     = 4
		local SCREAM_AT       = 2
		local SPHERE_MAX      = 22

		local RING_COLOR      = Color3.fromRGB(230, 180, 255)
		local SCREAM_TEXT     = "IM THE..... REAL REVENGE CLAWS!!!!"
		local SCREAM_SOUND    = ""
		local EXPLOSION_SOUND = "rbxassetid://4614478550"

			local char = lp2.Character
			return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"))
		end

		local function ultShake(duration, power)
			local char = lp2.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			if not hum then return end
			local start, conn = tick(), nil
			conn = RS2.RenderStepped:Connect(function()
				local t = tick() - start
				if t >= duration or not hum.Parent then
					hum.CameraOffset = Vector3.zero
					conn:Disconnect()
					return
				end
				local d = 1 - t / duration
				hum.CameraOffset = Vector3.new((math.random()-.5)*2*power*d, (math.random()-.5)*2*power*d, (math.random()-.5)*2*power*d)
			end)
		end

		local function makeNeonPart(color)
			local p = Instance.new("Part")
			p.Anchored = true
			p.CanCollide = false
			p.CanQuery = false
			p.CanTouch = false
			p.Massless = true
			p.Material = Enum.Material.Neon
			p.Color = color or CORE_COLOR
			p.Parent = workspace
			return p
		end

		local function ultimateExplosion(pos)
			local spart = Instance.new("Part")

			spart.Position = pos

			local snd = Instance.new("Sound")

			snd.PlaybackSpeed = 0.9

					Transparency = 1,

			local head = char:FindFirstChild("Head") or char:FindFirstChild("UpperTorso")

				game:GetService("Chat"):Chat(head, SCREAM_TEXT, Enum.ChatColor.Red)

		local function ultGetMotors(char)
			local t = char:FindFirstChild("Torso")
			local root = char:FindFirstChild("HumanoidRootPart")
			if not (t and root) then return nil end
			return {
				RootJoint = root:FindFirstChild("RootJoint"),
				Neck = t:FindFirstChild("Neck"),
				LS = t:FindFirstChild("Left Shoulder"),
				RS = t:FindFirstChild("Right Shoulder"),
				LH = t:FindFirstChild("Left Hip"),
				RH = t:FindFirstChild("Right Hip"),
			}
		end

		local function ultClerp(a, b, t)
			return a:Lerp(b, math.clamp(t, 0, 1))
		end

		local function ultFlyPose()
			return {
				RootJoint = CFrame.new(0, 0, 0) * CFrame.Angles(math.rad(-25), 0, math.rad(180)),
				Neck     = CFrame.new(0, 1, 0) * CFrame.Angles(math.rad(-30), math.rad(8), math.rad(180)),
				LS       = CFrame.new(-1.45, 0.4, 0.15) * CFrame.Angles(math.rad(45), math.rad(-15), math.rad(-115)),
				RS       = CFrame.new(1.45, 0.35, -0.1) * CFrame.Angles(math.rad(60), math.rad(20), math.rad(120)),
				LH       = CFrame.new(-1, -0.9, 0.15) * CFrame.Angles(math.rad(-40), math.rad(-15), math.rad(-25)),
				RH       = CFrame.new(1, -0.85, -0.1) * CFrame.Angles(math.rad(-30), math.rad(20), math.rad(30)),
			}
		end

		local function animateUltFlyUp(duration)
			local char = lp2.Character
			if not char then return end
			local m = ultGetMotors(char)
			if not m or not m.RootJoint then return end
			local target = ultFlyPose()
			local base = {}
			for name, motor in pairs(m) do
				if motor then base[name] = motor.C0 end
			end
			local start = tick()
			while tick() - start < duration do
				local alpha = math.clamp((tick() - start) / duration, 0, 1)
				local t = 1 - (1 - alpha) * (1 - alpha)
				if m.RootJoint and base.RootJoint then m.RootJoint.C0 = ultClerp(base.RootJoint, target.RootJoint, t) end
				if m.Neck and base.Neck then m.Neck.C0 = ultClerp(base.Neck, target.Neck, t) end
				if m.LS and base.LS then m.LS.C0 = ultClerp(base.LS, target.LS, t) end
				if m.RS and base.RS then m.RS.C0 = ultClerp(base.RS, target.RS, t) end
				if m.LH and base.LH then m.LH.C0 = ultClerp(base.LH, target.LH, t) end
				if m.RH and base.RH then m.RH.C0 = ultClerp(base.RH, target.RH, t) end
				task.wait()
			end
			if m.RootJoint then m.RootJoint.C0 = target.RootJoint end
			if m.Neck then m.Neck.C0 = target.Neck end
			if m.LS then m.LS.C0 = target.LS end
			if m.RS then m.RS.C0 = target.RS end
			if m.LH then m.LH.C0 = target.LH end
			if m.RH then m.RH.C0 = target.RH end
			return base
		end

		local function holdUltFlyPose(duration)
			local char = lp2.Character
			if not char then return end
			local m = ultGetMotors(char)
			if not m or not m.RootJoint then return end
			local target = ultFlyPose()
			local start = tick()
			while tick() - start < duration do
				local pulse = math.sin((tick() - start) * 8) * 0.02
				if m.RootJoint then
					m.RootJoint.C0 = target.RootJoint * CFrame.Angles(pulse, 0, 0)
				end
				if m.LS then m.LS.C0 = target.LS end
				if m.RS then m.RS.C0 = target.RS end
				if m.Neck then m.Neck.C0 = target.Neck end
				if m.LH then m.LH.C0 = target.LH end
				if m.RH then m.RH.C0 = target.RH end
				task.wait()
			end
		end

		local char = lp2.Character
		local root = getRootLocal()
		if not (char and root) then return end

		print("[ULT] ⬆️ Bay lên + pose CFrame...")

		local bv = Instance.new("BodyVelocity")
		bv.MaxForce = Vector3.new(1e9, 1e9, 1e9)
		bv.Velocity = Vector3.new(0, FLY_HEIGHT / FLY_TIME, 0)
		bv.Parent = root

		task.spawn(function()
			animateUltFlyUp(FLY_TIME)
		end)
		task.wait(FLY_TIME)
		bv.Velocity = Vector3.zero

		task.spawn(function()
			holdUltFlyPose(CHARGE_TIME)
		end)

		local folder = Instance.new("Model")
		folder.Name = "UltSphereFX"
		folder.Parent = char

		local RED_C  = Color3.fromRGB(255, 40, 40)
		local BLUE_C = Color3.fromRGB(40, 120, 255)
		local PURPLE = GLOW_COLOR
		local centerCF = CFrame.new(0, 4, -5)
		local purpleCF  = CFrame.new(0, 9, -5)
		local MERGE_TIME = math.min(1.4, CHARGE_TIME * 0.35)

		local function makeBall(color, size, trans)
			local p = Instance.new("Part")
			p.Shape = Enum.PartType.Ball
			p.Material = Enum.Material.Neon
			p.Color = color
			p.Size = Vector3.new(size, size, size)
			p.Transparency = trans or 0
			p.CanCollide = false
			p.Massless = true
			p.Parent = folder
			local w = Instance.new("Weld")
			w.Part0 = root
			w.Part1 = p
			w.Parent = p
			return p, w
		end

		local ORB_SIZE = 8
		local redOrb, redW = makeBall(RED_C, ORB_SIZE, 0.08)
		local blueOrb, blueW = makeBall(BLUE_C, ORB_SIZE, 0.08)
		redW.C0 = centerCF * CFrame.new(-12, 0, 0)
		blueW.C0 = centerCF * CFrame.new(12, 0, 0)

		local redLight = Instance.new("PointLight")
		redLight.Color = RED_C
		redLight.Brightness = 4
		redLight.Range = 18
		redLight.Parent = redOrb
		local blueLight = Instance.new("PointLight")
		blueLight.Color = BLUE_C
		blueLight.Brightness = 4
		blueLight.Range = 18
		blueLight.Parent = blueOrb

		local function orbParticles(parent, color)
			local pe = Instance.new("ParticleEmitter")
			pe.Color = ColorSequence.new(color)
			pe.LightEmission = 1
			pe.Size = NumberSequence.new(0.8, 0)
			pe.Transparency = NumberSequence.new(0.2, 1)
			pe.Lifetime = NumberRange.new(0.25, 0.5)
			pe.Rate = 80
			pe.Speed = NumberRange.new(3, 8)
			pe.SpreadAngle = Vector2.new(180, 180)
			pe.Parent = parent
			return pe

		while tick() - mergeStart < MERGE_TIME do
			local t = (tick() - mergeStart) / MERGE_TIME
			local ease = t * t * (3 - 2 * t)
			local side = 12 * (1 - ease)
			local spin = tick() * 3.5
			local bob = math.sin(tick() * 5) * 0.6
			redW.C0 = centerCF * CFrame.new(-side, bob, 0) * CFrame.Angles(0, spin, 0)
			blueW.C0 = centerCF * CFrame.new(side, -bob, 0) * CFrame.Angles(0, -spin, 0)
			local sz = ORB_SIZE + ease * 3
			redOrb.Size = Vector3.new(sz, sz, sz)
			blueOrb.Size = Vector3.new(sz, sz, sz)
			redLight.Brightness = 4 + ease * 6
			blueLight.Brightness = 4 + ease * 6

		redOrb:Destroy()
		blueOrb:Destroy()

		local core, coreWeld = makeBall(CORE_COLOR, 5, 0.05)
		coreWeld.C0 = purpleCF
		local glow, glowWeld = makeBall(PURPLE, 7, 0.5)
		glowWeld.C0 = purpleCF

		ring.Material = Enum.Material.Neon
		ring.Color = RING_COLOR
		ring.Transparency = 0.3

		ringWeld.Part1 = ring
		ringWeld.C0 = purpleCF * CFrame.Angles(0, 0, math.rad(90))
		ringWeld.Parent = ring
		ring.Parent = folder

		local pl = Instance.new("PointLight")
		pl.Color = PURPLE
		pl.Range = 40
		pl.Brightness = 4
		pl.Parent = core

		local pe = Instance.new("ParticleEmitter")
		pe.Color = ColorSequence.new(CORE_COLOR, PURPLE)
		pe.LightEmission = 1
		pe.Size = NumberSequence.new(1.2, 0)
		pe.Transparency = NumberSequence.new(0.1, 1)
		pe.Lifetime = NumberRange.new(0.3, 0.7)
		pe.Rate = 50  -- lag fix
		pe.Speed = NumberRange.new(4, 12)
		pe.SpreadAngle = Vector2.new(180, 180)
		pe.Parent = core

		local flash = makeNeonPart(PURPLE)
		flash.Shape = Enum.PartType.Ball
		flash.Size = Vector3.new(8, 8, 8)
		flash.Transparency = 0.15
		flash.CFrame = root.CFrame * purpleCF
		DB2:AddItem(flash, 0.6)
		TS2:Create(flash, TweenInfo.new(0.55, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
			Size = Vector3.new(28, 28, 28),
			Transparency = 1,
		}):Play()

		local spikes = {}
		local golden = math.pi * (3 - math.sqrt(5))
		for i = 1, SPIKE_COUNT do
			local y = 1 - (i - 1) / math.max(SPIKE_COUNT - 1, 1) * 2
			local r = math.sqrt(math.max(0, 1 - y * y))
			local theta = golden * i
			local dir = Vector3.new(math.cos(theta) * r, y, math.sin(theta) * r)
			local sp = Instance.new("Part")
			sp.Material = Enum.Material.Neon
			sp.Color = PURPLE
			sp.Transparency = 0.2
			sp.Size = Vector3.new(0.6, 0.6, 1)
			sp.CanCollide = false
			sp.Massless = true
			local sw = Instance.new("Weld")
			sw.Part0 = root
			sw.Part1 = sp
			sw.Parent = sp
			sp.Parent = folder
			spikes[i] = {part = sp, weld = sw, dir = dir}
		end

		local remain = math.max(0.2, CHARGE_TIME - MERGE_TIME)
		ultShake(remain, 0.8)
		local screamed = false
		local start = tick()
		while tick() - start < remain do
			local t = tick() - start
			local prog = t / remain
			local pulse = 1 + math.sin(t * 12) * 0.05
			local size = (4 + (SPHERE_MAX - 4) * prog) * pulse
			core.Size = Vector3.new(size, size, size)
			glow.Size = Vector3.new(size * 1.35, size * 1.35, size * 1.35)
			core.Color = CORE_COLOR:Lerp(PURPLE, prog * 0.55)
			pl.Brightness = 3 + 8 * prog
			pl.Range = 35 + 40 * prog
			ring.Size = Vector3.new(0.4, size * 1.9, size * 1.9)
			ringWeld.C0 = purpleCF * CFrame.Angles(math.rad(t * 60), math.rad(t * 90), math.rad(90))
			for _, s in ipairs(spikes) do
				local len = size * (0.8 + 0.4 * math.abs(math.sin(t * 5 + s.dir.X * 7)))
				s.part.Size = Vector3.new(0.6, 0.6, len)
				s.weld.C0 = purpleCF
					* CFrame.new(s.dir * (size / 2 + len / 2))
					* CFrame.lookAt(Vector3.zero, s.dir)
			end
			if (not screamed) and (MERGE_TIME + t) >= SCREAM_AT then
				screamed = true
				scream()
			end
			task.wait()
		end

		print("[ULT] ⬇️ SLAM DOWN!")
		folder:Destroy()
		bv.Velocity = Vector3.new(0, -SLAM_SPEED, 0)
		local params = RaycastParams.new()
		params.FilterDescendantsInstances = {char}
		params.FilterType = Enum.RaycastFilterType.Exclude
		local slamStart = tick()
		while tick() - slamStart < 2 do
			local hit = workspace:Raycast(root.Position, Vector3.new(0, -8, 0), params)
			if hit then break end
			task.wait()
		end
		bv:Destroy()

		local root2 = getRootLocal()
		if root2 then
			ultimateExplosion(root2.Position - Vector3.new(0, 3, 0))
			root2.CFrame = root2.CFrame + Vector3.new(0, 2, 0)
		end

		-- restore motors roughly to neutral after slam
		local m = ultGetMotors(char)
		if m then
			if m.RootJoint then m.RootJoint.C0 = CFrame.new(0, 0, 0) * CFrame.Angles(-math.pi/2, 0, math.pi) end
			if m.Neck then m.Neck.C0 = CFrame.new(0, 1, 0) * CFrame.Angles(-math.pi/2, 0, math.pi) end
		end
	end)

	if not ran then
		warn("[P ULT] ERROR: " .. tostring(err))
	end

	if anim then anim.Disabled = false end
	Humanoid.WalkSpeed = 30
	attack = false
	print("[P] Real Revenge Claws V2 done")
end

Footsteps:Stop()
Footsteps2:Stop()
local Fire = it("Sound",Character.Torso)
Fire.SoundId = "rbxassetid://192104941"
Fire.Looped = true
